\section*{Research Overview}
\textbf{Reliable, compute-efficient factuality verification for LLMs:} Conformal factuality and risk control for RAG under distribution shift; masked-diffusion and lightweight entailment verifiers; agent evaluation with LLM and VLM judges. Related threads: confidence scores for auto-labeling, human feedback, and pluralistic alignment.\par
\textbf{Publication record:} Seven conference papers at NeurIPS, ICML, ICLR, WWW (oral), AISTATS, UAI, and HCOMP\@; first or co-first author on three (ICML, WWW, HCOMP); plus a first-author 2026 preprint.

\section*{Research Experience}
\entry{Conformal Factuality for RAG-based LLMs: Robustness and Efficiency Analysis}{2025--2026}
\begin{itemize}
\item \textit{arXiv 2026 (first author); ICML Workshop 2025:} Led a study across three benchmarks and multiple model families; found that lightweight entailment verifiers match or outperform LLM confidence scorers while requiring over 100$\times$ fewer FLOPs.
\item Introduced informativeness-aware metrics; diagnosed vacuous outputs at high factuality levels and the fragility of conformal filtering under distribution shifts and distractors.
\end{itemize}
\entry{Palimpsest: Bidirectional Scoring for Span-Level Hallucination Detection}{2026--present}
\begin{itemize}
\item Proposed a masked-diffusion verifier that masks entity spans and scores length-normalized token recovery; designed reference-grounded and reference-free evaluations against trained checkers and token-probability baselines.
\item Designed a matched cloze comparison of a masked-diffusion language model (Dream-7B) and an autoregressive LLM (Qwen2.5-7B) to isolate reconstruction-based verification behavior; ongoing work extends the approach to conformal factuality under generator shift.
\end{itemize}
\entry{Confidence Scores for Reliable Data Annotation and Pseudolabeling}{2024--2025}
\begin{itemize}
\item \textit{NeurIPS 2024:} Co-developed confidence functions and thresholding methods for automated labeling under explicit error constraints, improving coverage by up to 60\% over calibration baselines while holding error below 5\%.
\item \textit{ICML 2025 (equal contribution):} Connected automated-label quality and quantity through a coverage--error objective in pseudolabeling-based semi-supervised learning.
\end{itemize}
\entry{Human Feedback, Preference Learning, and Pluralistic Alignment}{2023--2025}
\begin{itemize}
\item \textit{ICLR 2025:} Co-developed PAL, a pluralistic alignment framework that learns shared structure and individual differences from heterogeneous preferences.
\item \textit{WWW 2025 (oral):} Designed crowdsourced clustering studies of cognitive overload and contextual bias to understand how question format changes annotation quality.
\item \textit{AISTATS 2024; UAI 2025:} Studied identifiability and recovery of preference distributions from limited pairwise comparisons, and nonlinear metric learning in an RKHS, respectively.
\item \textit{HCOMP 2023:} Developed active querying for crowdsourced clustering with theoretical recovery guarantees and evaluation on human-generated data.
\end{itemize}
